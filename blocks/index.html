Pürland 首頁程式碼

貼進 Shopline 後台用。按綠色按鈕複製，不用打開任何檔案。

怎麼用

  1. 按下面的 「複製這段程式碼」（按鈕變綠色就是複製好了）
  2. 到 Shopline 後台 →「網店設計」→ 首頁
  3. 按 「+ 元件」 → 選 「客製化語法元件」
  4. 點進「語法」那個框，按 Cmd + V 貼上
  5. 按 「儲存」，然後用右上角 「預覽商店」 看效果
先不要按「發佈」。儲存後先預覽確認沒問題，再決定要不要發佈上線。

<script>
(function(){
  var root=document.querySelector('.plx-hero'); if(!root) return;
  var box=root.closest('.plx'); if(box) box.classList.add('plx-js');
  var car=root.querySelector('.pl-hero-carousel'); if(!car) return;
  var slides=[].slice.call(car.querySelectorAll('.pl-hero-slide'));
  var dots=[].slice.call(car.querySelectorAll('.pl-hero-dot'));
  var prev=car.querySelector('.pl-hero-prev'), next=car.querySelector('.pl-hero-next');
  var i=0,timer=null,manual=false;
  function show(n){ i=(n+slides.length)%slides.length;
    slides.forEach(function(s,k){ var a=k===i; s.classList.toggle('is-active',a); s.setAttribute('aria-hidden',String(!a)); });
    dots.forEach(function(d,k){ var a=k===i; d.classList.toggle('is-active',a); d.setAttribute('aria-selected',String(a)); }); }
  function stop(){ if(timer){clearInterval(timer); timer=null;} }
  function start(){ if(timer||manual) return; car.classList.remove('is-paused'); timer=setInterval(function(){show(i+1);},5000); }
  function pause(){ if(timer){ stop(); car.classList.add('is-paused'); } }
  function take(){ manual=true; stop(); car.classList.add('is-paused'); }
  car.addEventListener('mouseenter',pause); car.addEventListener('mouseleave',start);
  car.addEventListener('focusin',pause); car.addEventListener('focusout',start);
  if(prev) prev.addEventListener('click',function(){take();show(i-1);});
  if(next) next.addEventListener('click',function(){take();show(i+1);});
  dots.forEach(function(d,k){ d.addEventListener('click',function(){take();show(k);}); });
  show(0); start();
  var io=new IntersectionObserver(function(es){ es.forEach(function(e){ if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target);} }); },{threshold:.05,rootMargin:'0px 0px -8% 0px'});
  root.querySelectorAll('.reveal').forEach(function(el){io.observe(el);});
  setTimeout(function(){ root.querySelectorAll('.reveal:not(.in)').forEach(function(el){el.classList.add('in');}); },2500);
})();
</script>
